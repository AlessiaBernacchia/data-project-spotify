# count NaN for each column
count_na_per_column <- function(data) {
  sapply(data, function(col) sum(is.na(col)))}

count_na_per_column_df <- function(data) {
  na_counts <- sapply(data, function(col) sum(is.na(col)))
  na_counts_df <- data.frame(
    feature = names(na_counts),
    na_count = na_counts
  )
  na_counts_df %>% arrange(desc(na_count))
}

# extract first genre for the spotify genre
extract_first_genre <- function(df, genre_column = "spotify_genre") {
  # check if column exists
  if (!genre_column %in% colnames(df)) {
    stop(paste("La colonna", genre_column, "non esiste nel dataframe"))
  }
  
  # removes useless not required characters
  df[[genre_column]] <- gsub("\\[|\\]|'", "", df[[genre_column]])
  
  # create a list, with div = ,
  genre_list <- strsplit(df[[genre_column]], split = ",")
  
  # extract only the first genre for each line or rename "others"
  df$first_genre <- sapply(genre_list, function(x) {
    x <- trimws(x)  # remove blanks
    if (length(x) == 0 || is.na(x[1]) || x[1] == "") {
      return("others")
    } else {
      return(x[1])
    }
  })
  
  # convert to categorical value
  df$first_genre <- as.factor(df$first_genre)
  
  # output
  return(df)
}

# process and encode the genre column
process_and_encode_genre <- function(df, genre_column = "first_genre", top_n = 15) {
  # check if column exists
  if (!genre_column %in% colnames(df)) {
    stop(paste("La colonna", genre_column, "non esiste nel dataframe"))
  }
  
  # count the frequency
  genre_counts <- table(df[[genre_column]])
  
  # put them in decrescent order
  genre_counts_sorted <- sort(genre_counts, decreasing = TRUE)
  
  # take top N genres
  top_genres <- names(genre_counts_sorted)[1:min(top_n, length(genre_counts_sorted))]
  
  # rename not in top N into 'others'
  df[[genre_column]] <- ifelse(
    df[[genre_column]] %in% top_genres,
    df[[genre_column]],
    "others"
  )
  
  # convert
  df[[genre_column]] <- as.factor(df[[genre_column]])
  
  # remove original column
  if ("spotify_genre" %in% colnames(df)) {
    df <- subset(df, select = -c(spotify_genre))
  }
  
  # create dummy variables (one-hot encoding)
  dummy <- dummyVars(paste("~", genre_column), data = df)
  dummy_df <- data.frame(predict(dummy, newdata = df))
  
  final_df <- bind_cols(
    df %>% select(-all_of(genre_column)),
    dummy_df
  )
  
  # output
  return(final_df)
}

# encode popularity column
encode_spotify_popularity <- function(df, column = "spotify_track_popularity") {
  # check if column exists
  if (!column %in% colnames(df)) {
    stop(paste("La colonna", column, "non esiste nel dataframe"))
  }
  
  categorize_popularity <- function(x) {
    if (is.na(x)) return(NA)
    if (x <= 25) {
      return("low")
    } else if (x <= 50) {
      return("medium")
    } else if (x <= 75) {
      return("high")
    } else if (x <= 100) {
      return("very high")
    } else {
      return(NA)  # out of range
    }
  }
  
  df[[column]] <- sapply(df[[column]], categorize_popularity)
  
  # convert in ordered factor
  df[[column]] <- factor(df[[column]], 
                         levels = c("low", "medium", "high", "very high"), 
                         ordered = TRUE)
  
  # output
  return(df)
}

# plots the confusion matrix
plot_conf_matrix <- function(conf_matrix_object, title = "Confusion Matrix") {
  
  # converts confusion matrix into dataframe
  cm_df <- as.data.frame(conf_matrix_object$table)
  
  # find columns names (Truth, Prediction, Freq)
  colnames(cm_df) <- tolower(colnames(cm_df)) # rende tutto minuscolo
  
  # rename them
  names(cm_df)[names(cm_df) %in% c("reference", "truth")] <- "Actual"
  names(cm_df)[names(cm_df) == "prediction"] <- "Predicted"
  names(cm_df)[names(cm_df) %in% c("freq", "n")] <- "Freq"
  
  # plot
  p <- ggplot(cm_df, aes(x = Predicted, y = Actual, fill = Freq)) +
    geom_tile(color = "white", linewidth = 0.5) +
    geom_text(aes(label = Freq), size = 5, color = "black") +
    scale_fill_gradient(low = "white", high = "steelblue") +
    labs(title = title,
         x = "Predicted Class",
         y = "Actual Class",
         fill = "Count") +
    theme_minimal(base_size = 14) +
    theme(plot.title = element_text(hjust = 0.5, face = "bold"))
  
  return(p)
}
