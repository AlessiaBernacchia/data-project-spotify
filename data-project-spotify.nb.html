set.seed(42)
full_data_clean$top10 <- as.factor(full_data_clean$top10)
spotify_train_test <- full_data_clean |> initial_split(prop =  0.7)
spotify_train <- training(spotify_train_test); 
spotify_test  <- testing(spotify_train_test)
rand_forest_spec <- 
  rand_forest(mtry = 5, trees = 200) |>  
  set_engine('randomForest') |> 
  set_mode('classification')

rf_recipe = recipe(top10~.,
                   data=spotify_train)

rf_wf <- workflow() |> 
  add_recipe(rf_recipe) |> 
  add_model(rand_forest_spec)

rf_wf_fit <- rf_wf |> 
  fit(spotify_train)
